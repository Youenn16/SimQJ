cmake_minimum_required(VERSION 3.15)

project(simQJ LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Source files
set(SOURCE_FILES
    src/math.cpp
    src/output_data.cpp
    src/simulation.cpp
    src/models/ising_model.cpp
)

# Core library
add_library(core ${SOURCE_FILES})

# Main executable
add_executable(simQJ src/main.cpp)
target_link_libraries(simQJ PRIVATE core)

# BLAS + LAPACK
find_package(BLAS REQUIRED)
find_package(LAPACK REQUIRED) 

set(LAPACKE_INCLUDE_DIR "/opt/homebrew/opt/lapack/include")
set(LAPACKE_LIB "/opt/homebrew/opt/lapack/lib/liblapacke.dylib")

target_include_directories(core
    PUBLIC
        ${PROJECT_SOURCE_DIR}/include
        ${LAPACKE_INCLUDE_DIR}
)

target_link_libraries(core
    PUBLIC
        BLAS::BLAS
        LAPACK::LAPACK
        ${LAPACKE_LIB}
)

target_include_directories(core
    PUBLIC
        ${PROJECT_SOURCE_DIR}/include
    SYSTEM PUBLIC
        ${LAPACKE_INCLUDE_DIR}
)

target_link_libraries(core
    PUBLIC
        BLAS::BLAS
        LAPACK::LAPACK
        ${LAPACKE_LIB}
)

if(APPLE)
    target_compile_options(core PRIVATE -isysroot ${CMAKE_OSX_SYSROOT} -D_LIBCPP_DISABLE_AVAILABILITY)
    target_include_directories(core SYSTEM PUBLIC ${LAPACKE_INCLUDE_DIR})
endif()

include(FetchContent)

FetchContent_Declare(
    Catch2
    GIT_REPOSITORY https://github.com/catchorg/Catch2.git
    GIT_TAG v3.5.2
)

FetchContent_MakeAvailable(Catch2)

include(${Catch2_SOURCE_DIR}/extras/Catch.cmake)

enable_testing()
add_subdirectory(tests)
